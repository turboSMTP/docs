The turboSMTP REST API works with any language or tool that can make HTTP requests. Each example below sends a POST request to the global send endpoint with `consumerKey` and `consumerSecret` headers and a JSON body containing the standard message fields. Swap in your own credentials and addresses, then run the code to send your first email.

<Tip>
  Replace `YOUR_CONSUMER_KEY` and `YOUR_CONSUMER_SECRET` with the keys from your dashboard at **Settings → Integrations → API Keys**. Your Consumer Secret is shown only once at creation time, so store it securely.
</Tip>

<CodeGroup>

```bash cURL
curl -X POST 'https://api.turbo-smtp.com/api/v2/mail/send' \
  -H 'consumerKey: YOUR_CONSUMER_KEY' \
  -H 'consumerSecret: YOUR_CONSUMER_SECRET' \
  -H 'Content-Type: application/json' \
  -d '{
    "from": "hello@your-company.com",
    "to": "recipient@example.com",
    "subject": "Hello from turboSMTP",
    "content": "This is a plain text message.",
    "html_content": "<p>This is an <strong>HTML</strong> message.</p>"
  }'
```

```python Python
import requests

url = 'https://api.turbo-smtp.com/api/v2/mail/send'
headers = {
    'consumerKey': 'YOUR_CONSUMER_KEY',
    'consumerSecret': 'YOUR_CONSUMER_SECRET',
    'Content-Type': 'application/json'
}
payload = {
    'from': 'hello@your-company.com',
    'to': 'recipient@example.com',
    'subject': 'Hello from turboSMTP',
    'content': 'This is a plain text message.',
    'html_content': '<p>This is an <strong>HTML</strong> message.</p>'
}
response = requests.post(url, headers=headers, json=payload, timeout=30)
print(response.status_code, response.json())
```


```go Go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
    "log"
    "time"
)

func main() {
    payload := map[string]string{
        "from":         "hello@your-company.com",
        "to":           "recipient@example.com",
        "subject":      "Hello from turboSMTP",
        "content":      "This is a plain text message.",
        "html_content": "<p>This is an <strong>HTML</strong> message.</p>",
    }
    body, err := json.Marshal(payload)
    if err != nil {
        log.Fatal(err)
    }
    req, err := http.NewRequest("POST", "https://api.turbo-smtp.com/api/v2/mail/send", bytes.NewBuffer(body))
    if err != nil {
        log.Fatal(err)
    }
    req.Header.Set("consumerKey", "YOUR_CONSUMER_KEY")
    req.Header.Set("consumerSecret", "YOUR_CONSUMER_SECRET")
    req.Header.Set("Content-Type", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    resp, err := client.Do(req)
    if err != nil {
        log.Fatal(err)
    }
    defer resp.Body.Close()
    fmt.Println(resp.Status)
}
```

```java Java
import java.net.http.*;
import java.net.URI;
import java.nio.charset.StandardCharsets;

public class SendEmail {
    public static void main(String[] args) throws Exception {
        String json = "{\"from\":\"hello@your-company.com\"," +
            "\"to\":\"recipient@example.com\"," +
            "\"subject\":\"Hello from turboSMTP\"," +
            "\"content\":\"This is a plain text message.\"," +
            "\"html_content\":\"<p>This is an <strong>HTML</strong> message.</p>\"}";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://api.turbo-smtp.com/api/v2/mail/send"))
            .header("consumerKey", "YOUR_CONSUMER_KEY")
            .header("consumerSecret", "YOUR_CONSUMER_SECRET")
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(json, StandardCharsets.UTF_8))
            .build();
        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
        System.out.println(response.body());
    }
}
```

## Before you run the examples

Use credentials with **HTTP API Sending** enabled and a sender domain configured for your account. Python uses the `requests` package (`python -m pip install requests`); the Node.js example uses `axios` (`npm install axios`). PHP requires its cURL extension, and the Java example requires Java 11 or later. The C# example uses top-level statements in a modern .NET console project.

To send through turboSMTP's European infrastructure, replace `api.turbo-smtp.com` with `api.eu.turbo-smtp.com`. turboSMTP is an Italian company with European servers; see [hosts and data residency](/api-reference/base-urls).
